    // WebSocket Connection
    const socket = new WebSocket("wss://yourserver.com/challenge");

    socket.onmessage = (event) => {
      const data = JSON.parse(event.data);

      if (data.type === "updateScore") {
        updateBar(data.red, data.blue);
      }

      if (data.type === "resetTimer") {
        resetTimer(data.time);
      }
    };

    function updateBar(redScore, blueScore) {
      document.getElementById("redBar").style.width = `${redScore}%`;
      document.getElementById("blueBar").style.width = `${blueScore}%`;
    }

    // Countdown Timer
    let timer = 180; // 3 minutes
    const timerEl = document.getElementById("timer");
    const timerInterval = setInterval(() => {
      if (timer > 0) {
        timer--;
        timerEl.textContent = formatTime(timer);
      }
    }, 1000);

    function formatTime(seconds) {
      const m = Math.floor(seconds / 60);
      const s = seconds % 60;
      return `${m}:${s < 10 ? '0' : ''}${s}`;
    }

    function resetTimer(seconds) {
      timer = seconds;
    }
